**Tech Stack:** Rust (edition 2024), Ratatui 0.29, Crossterm 0.28, Rand 0.8

```rust
#[derive(Clone, Copy)]
pub struct Room {
    pub x: usize,
    pub y: usize,
    pub w: usize,
    pub h: usize,
}

impl Room {
    pub fn new(x: usize, y: usize, w: usize, h: usize) -> Self {
        Room { x, y, w, h }
    }

    pub fn center(&self) -> (usize, usize) {
        (self.x + self.w / 2, self.y + self.h / 2)
    }

    pub fn overlaps(&self, other: &Room) -> bool {
        self.x < other.x + other.w + 1
            && self.x + self.w + 1 > other.x
            && self.y < other.y + other.h + 1
            && self.y + self.h + 1 > other.y
    }
}
```

```rust
use rand::Rng;

pub fn generate(rng: &mut impl Rng) -> Map {
    let mut map = Map::new();
    let num_rooms = rng.gen_range(6..=10);

    for _ in 0..50 {
        if map.rooms.len() >= num_rooms {
            break;
        }
        let w = rng.gen_range(4..=10);
        let h = rng.gen_range(4..=8);
        let x = rng.gen_range(1..MAP_WIDTH - w - 1);
        let y = rng.gen_range(1..MAP_HEIGHT - h - 1);
        let room = Room::new(x, y, w, h);

        if map.rooms.iter().any(|r| r.overlaps(&room)) {
            continue;
        }

        // Carve room tiles
        for ry in room.y..room.y + room.h {
            for rx in room.x..room.x + room.w {
                map.set(rx, ry, Tile::Floor);
            }
        }

        // Connect to previous room with L-shaped corridor
        if let Some(prev) = map.rooms.last() {
            let (px, py) = prev.center();
            let (cx, cy) = room.center();
            carve_h_corridor(&mut map, px, cx, py);
            carve_v_corridor(&mut map, py, cy, cx);
        }

        map.rooms.push(room);
    }

    map
}

fn carve_h_corridor(map: &mut Map, x1: usize, x2: usize, y: usize) {
    let (start, end) = if x1 < x2 { (x1, x2) } else { (x2, x1) };
    for x in start..=end {
        if map.get(x, y) == Tile::Wall {
            map.set(x, y, Tile::Corridor);
        }
    }
}

fn carve_v_corridor(map: &mut Map, y1: usize, y2: usize, x: usize) {
    let (start, end) = if y1 < y2 { (y1, y2) } else { (y2, y1) };
    for y in start..=end {
        if map.get(x, y) == Tile::Wall {
            map.set(x, y, Tile::Corridor);
        }
    }
}
```

```rust
use crate::entity::{Monster, Player};

pub fn resolve_player_attack(
    player: &mut Player,
    monster: &mut Monster,
    rng: &mut impl Rng,
) -> String {
    let hit_roll = roll_d6(rng);
    if hit_roll < 4 {
        return format!("You missed the {}!", monster.kind.name());
    }

    let damage = roll_d6(rng);
    monster.hp -= damage;

    if monster.hp <= 0 {
        monster.alive = false;
        player.max_hp += 1;
        player.hp += 1;
        return format!(
            "You slew the {}! +1 max HP (now {}/{})",
            monster.kind.name(), player.hp, player.max_hp
        );
    }

    format!(
        "You hit the {} for {} damage! ({} HP left)",
        monster.kind.name(), damage, monster.hp
    )
}
```

```rust
pub fn resolve_monster_attack(
    monster: &Monster,
    player: &mut Player,
    rng: &mut impl Rng,
) -> (String, bool) {
    let hit_roll = roll_d6(rng);
    if hit_roll < 4 {
        return (format!("The {} missed you!", monster.kind.name()), false);
    }

    let damage = roll_d6(rng);
    player.hp -= damage;

    let died = player.hp <= 0;
    let msg = format!(

        monster.kind.name(), damage, player.hp.max(0)
    );
    (msg, died)
}
```

```rust
fn spawn_monsters(map: &Map, rng: &mut impl Rng) -> Vec<Monster> {
    let rooms = &map.rooms;
    let kinds = [MonsterKind::Goblin, MonsterKind::Mage, MonsterKind::Rat];
    let mut monsters = Vec::new();

    for (i, &kind) in kinds.iter().enumerate() {
        // Pick a room that isn't room 0 (player start), cycle through available rooms
        let room_idx = if rooms.len() > 1 {
            let idx = (i % (rooms.len() - 1)) + 1;
            idx
        } else {
            0
        };
        let room = &rooms[room_idx];
        let mx = rng.gen_range(room.x..room.x + room.w);
        let my = rng.gen_range(room.y..room.y + room.h);
        monsters.push(Monster::new(mx, my, kind));
    }
    monsters
}
```

```rust
fn spawn_potions(map: &mut Map, rng: &mut impl Rng, monsters: &[Monster]) {
    let count = {
        let roll: f32 = rng.gen();
        if roll < 0.30 { 2 } else if roll < 0.60 { 1 } else { 0 }
    };
    let mut placed = 0;
    let mut attempts = 0;
    while placed < count && attempts < 200 {
        attempts += 1;
        let x = rng.gen_range(1..MAP_WIDTH - 1);
        let y = rng.gen_range(1..MAP_HEIGHT - 1);
        if !map.is_walkable(x, y) {
            continue;
        }
        if monsters.iter().any(|m| m.x == x && m.y == y) {
            continue;
        }
        map.set(x, y, Tile::Potion);
        placed += 1;
    }
}
```

```rust
    pub fn try_spawn_stair(&mut self) {
        if self.monsters.iter().any(|m| m.alive) {
            return; // not all dead yet
        }
        // Find a walkable tile that isn't the player's position
        let mut candidates: Vec<(usize, usize)> = Vec::new();
        for y in 0..MAP_HEIGHT {
            for x in 0..MAP_WIDTH {
                if self.map.is_walkable(x, y)
                    && !(x == self.player.x && y == self.player.y)
                    && self.map.get(x, y) != Tile::Stair
                    && self.map.get(x, y) != Tile::Potion
                {
                    candidates.push((x, y));
                }
            }
        }
        let (sx, sy) = if candidates.is_empty() {
            (self.player.x, self.player.y) // emergency fallback
        } else {
            candidates[self.rng.gen_range(0..candidates.len())]
        };
        self.map.set(sx, sy, Tile::Stair);
        self.status_msg = String::from("The way down is revealed...");
    }
```

- [ ] **Step 7: Implement `Game::move_player()` — handles input-driven movement**

This resolves the full tile interaction priority: monster → potion → stair → walk.

```rust
    pub fn move_player(&mut self, dx: i32, dy: i32) {
        let nx = self.player.x as i32 + dx;
        let ny = self.player.y as i32 + dy;

        if nx < 0 || ny < 0 || nx >= MAP_WIDTH as i32 || ny >= MAP_HEIGHT as i32 {
            return;
        }
        let nx = nx as usize;
        let ny = ny as usize;

        if !self.map.is_walkable(nx, ny) {
            return;
        }

        // Priority 1: monster combat
        if let Some(idx) = self.monsters.iter().position(|m| m.alive && m.x == nx && m.y == ny) {
            let msg = crate::combat::resolve_player_attack(
                &mut self.player,
                &mut self.monsters[idx],
                &mut self.rng,
            );
            self.status_msg = msg;

            // Monster counter-attacks if still alive
            if self.monsters[idx].alive {
                let (msg2, died) = crate::combat::resolve_monster_attack(
                    &self.monsters[idx],
                    &mut self.player,
                    &mut self.rng,
                );
                self.status_msg = msg2;
                if died {
                    self.state = GameState::GameOver;
                    return;
                }
            } else {
                // Monster just died — check if stair should appear
                self.try_spawn_stair();
            }
            return; // player doesn't move into monster tile
        }

        // Move player
        self.player.x = nx;
        self.player.y = ny;

        // Priority 2: potion
        if self.map.get(nx, ny) == Tile::Potion {
            self.player.hp = self.player.max_hp;
            self.map.set(nx, ny, Tile::Floor);
            self.status_msg = format!(
                "You drank a potion and feel restored! ({} HP)",
                self.player.hp
            );
            return;
        }

        // Priority 3: staircase
        if self.map.get(nx, ny) == Tile::Stair {
            if self.level == 10 {
                self.state = GameState::Victory;
            } else {
                self.status_msg = String::from("The stairs lead deeper into the dungeon...");
                self.next_level();
            }
            return;
        }
    }
```

- [ ] **Step 8: Implement `Game::move_monsters()` — chase AI for all living monsters**

```rust
    pub fn move_monsters(&mut self) {
        let px = self.player.x;
        let py = self.player.y;

        for i in 0..self.monsters.len() {
            if !self.monsters[i].alive {
                continue;
            }

            let mx = self.monsters[i].x as i32;
            let my = self.monsters[i].y as i32;
            let dx = px as i32 - mx;
            let dy = py as i32 - my;

            // Determine step direction — prefer axis with greater distance; horizontal on tie
            let (step_x, step_y) = if dx.abs() >= dy.abs() {
                (dx.signum(), 0)
            } else {
                (0, dy.signum())
            };

            let nx = (mx + step_x) as usize;
            let ny = (my + step_y) as usize;

            // Blocked by wall or another monster
            if !self.map.is_walkable(nx, ny) {
                continue;
            }
            if self.monsters.iter().any(|m| m.alive && m.x == nx && m.y == ny) {
                continue;
            }

            // Check if stepping onto player
            if nx == px && ny == py {
                let (msg, died) = crate::combat::resolve_monster_attack(
                    &self.monsters[i],
                    &mut self.player,
                    &mut self.rng,
                );
                self.status_msg = msg;
                if died {
                    self.state = GameState::GameOver;
                    return;
                }
            } else {
                self.monsters[i].x = nx;
                self.monsters[i].y = ny;
            }
        }
    }
```

```rust
fn render_dungeon(frame: &mut Frame, game: &Game, area: Rect) {
    let mut lines: Vec<Line> = Vec::new();

    for y in 0..MAP_HEIGHT {
        let mut spans: Vec<Span> = Vec::new();
        for x in 0..MAP_WIDTH {
            // Determine what to render at this cell (priority: player > monster > potion > stair > tile)
            if game.player.x == x && game.player.y == y {
                spans.push(Span::styled("@", Style::default().fg(Color::White).add_modifier(Modifier::BOLD)));
                continue;
            }

            if let Some(m) = game.monsters.iter().find(|m| m.alive && m.x == x && m.y == y) {
                let color = match m.kind {
                    MonsterKind::Goblin => Color::Blue,
                    MonsterKind::Mage => Color::Red,
                    MonsterKind::Rat => Color::DarkGray, // closest to brown
                };
                spans.push(Span::styled(m.kind.char().to_string(), Style::default().fg(color).add_modifier(Modifier::BOLD)));
                continue;
            }

            let (ch, style) = match game.map.get(x, y) {
                Tile::Wall => ('#', Style::default().fg(Color::DarkGray)),
                Tile::Floor | Tile::Corridor => ('.', Style::default().fg(Color::Black).bg(Color::Black)),
                Tile::Stair => ('>', Style::default().fg(Color::Yellow)),
                Tile::Potion => ('P', Style::default().fg(Color::Green)),
            };
            spans.push(Span::styled(ch.to_string(), style));
        }
        lines.push(Line::from(spans));
    }

    let para = Paragraph::new(lines)
        .block(Block::default().borders(Borders::NONE))
        .style(Style::default().bg(Color::Black));
    frame.render_widget(para, area);
}
```

```rust
fn render_overlay(frame: &mut Frame, area: Rect, title: &str, subtitle: &str, color: Color) {
    // Darken background
    frame.render_widget(Clear, area);

    let block = Block::default()
        .borders(Borders::ALL)
        .border_style(Style::default().fg(color))
        .style(Style::default().bg(Color::Black));

    // Center a popup box
    let popup_area = centered_rect(50, 20, area);
    frame.render_widget(block, popup_area);

    let inner = Rect {
        x: popup_area.x + 2,
        y: popup_area.y + popup_area.height / 2 - 1,
        width: popup_area.width.saturating_sub(4),
        height: 3,
    };

    let lines = vec![
        Line::from(Span::styled(
            title,
            Style::default().fg(color).add_modifier(Modifier::BOLD),
        )),
        Line::from(Span::raw(subtitle)),
        Line::from(Span::styled(
            "Press any key to exit.",
            Style::default().fg(Color::Gray),
        )),
    ];

    let para = Paragraph::new(lines).alignment(Alignment::Center);
    frame.render_widget(para, inner);
}

fn centered_rect(percent_x: u16, percent_y: u16, r: Rect) -> Rect {
    let popup_layout = Layout::default()
        .direction(Direction::Vertical)
        .constraints([
            Constraint::Percentage((100 - percent_y) / 2),
            Constraint::Percentage(percent_y),
            Constraint::Percentage((100 - percent_y) / 2),
        ])
        .split(r);

    Layout::default()
        .direction(Direction::Horizontal)
        .constraints([
            Constraint::Percentage((100 - percent_x) / 2),
            Constraint::Percentage(percent_x),
            Constraint::Percentage((100 - percent_x) / 2),
        ])
        .split(popup_layout[1])[1]
}
```

```rust
fn main() -> Result<(), io::Error> {
    // Size check before entering raw mode
    if let Err(msg) = check_terminal_size() {
        eprintln!("Error: {}", msg);
        std::process::exit(1);
    }

    // Setup terminal
    enable_raw_mode()?;
    let mut stdout = io::stdout();
    execute!(stdout, EnterAlternateScreen, EnableMouseCapture)?;
    let backend = CrosstermBackend::new(stdout);
    let mut terminal = Terminal::new(backend)?;

    let mut game = Game::new();

    let result = run_loop(&mut terminal, &mut game);

    // Teardown terminal (always, even on error)
    disable_raw_mode()?;
    execute!(
        terminal.backend_mut(),
        LeaveAlternateScreen,
        DisableMouseCapture
    )?;
    terminal.show_cursor()?;

    result
}
```

- [ ] **Step 4: Implement `run_loop()`**

```rust
fn run_loop(
    terminal: &mut Terminal<CrosstermBackend<io::Stdout>>,
    game: &mut Game,
) -> Result<(), io::Error> {
    loop {
        terminal.draw(|frame| render::render_game(frame, game))?;

        if let Event::Key(key) = event::read()? {
            match game.state {
                GameState::Playing => match key.code {
                    KeyCode::Esc => break,
                    KeyCode::Up => {
                        game.move_player(0, -1);
                        if game.state == GameState::Playing {
                            game.move_monsters();
                        }
                    }
                    KeyCode::Down => {
                        game.move_player(0, 1);
                        if game.state == GameState::Playing {
                            game.move_monsters();
                        }
                    }
                    KeyCode::Left => {
                        game.move_player(-1, 0);
                        if game.state == GameState::Playing {
                            game.move_monsters();
                        }
                    }
                    KeyCode::Right => {
                        game.move_player(1, 0);
                        if game.state == GameState::Playing {
                            game.move_monsters();
                        }
                    }
                    _ => {}
                },
                GameState::GameOver | GameState::Victory => break,
            }
        }
    }
    Ok(())
}
```
